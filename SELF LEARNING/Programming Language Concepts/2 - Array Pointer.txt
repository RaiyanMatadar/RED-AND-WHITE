Array Pointer

#include <stdio.h>

int main() {
    int a = 10, b = 20, c = 30;
    int *arr[3];  // array of 3 int pointers

    arr[0] = &a;  // store address of a
    arr[1] = &b;  // store address of b
    arr[2] = &c;  // store address of c

    // accessing values using dereferencing
    printf("%d\n", *arr[0]);  // prints 10
    printf("%d\n", *arr[1]);  // prints 20
    printf("%d\n", *arr[2]);  // prints 30

    return 0;
}
